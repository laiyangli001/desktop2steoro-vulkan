cmake_minimum_required(VERSION 3.21)
project(d2s_desktop_duplication LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "d2s_desktop_duplication is currently implemented for Windows only")
endif()
add_library(d2s_desktop_duplication SHARED desktop_duplication.cpp)
target_compile_features(d2s_desktop_duplication PRIVATE cxx_std_17)
target_compile_definitions(d2s_desktop_duplication PRIVATE D2S_DESKTOP_DUPLICATION_BUILD=1 WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(d2s_desktop_duplication PRIVATE "\${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(d2s_desktop_duplication PRIVATE d3d11 dxgi)
set_target_properties(d2s_desktop_duplication PROPERTIES OUTPUT_NAME "d2s_desktop_duplication" PREFIX "")
