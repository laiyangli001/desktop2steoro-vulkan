cmake_minimum_required(VERSION 3.21)
project(d2s_amd_encoder LANGUAGES CXX)

if(NOT WIN32)
    message(FATAL_ERROR "d2s_amd_encoder is currently implemented for Windows only")
endif()

add_library(d2s_amd_encoder SHARED amd_encoder.cpp)
target_compile_features(d2s_amd_encoder PRIVATE cxx_std_17)
target_compile_definitions(d2s_amd_encoder PRIVATE D2S_AMD_ENCODER_BUILD=1 WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_link_libraries(d2s_amd_encoder PRIVATE d3d11 dxgi)
if(MSVC)
    target_compile_options(d2s_amd_encoder PRIVATE /FIwinternl.h)
endif()

set(D2S_AMF_ROOT "" CACHE PATH "AMF SDK root containing amf/public/include")
if(D2S_AMF_ROOT AND EXISTS "${D2S_AMF_ROOT}/amf/public/include/core/Context.h")
    target_compile_definitions(d2s_amd_encoder PRIVATE D2S_AMF_SDK=1)
    target_include_directories(d2s_amd_encoder PRIVATE "${D2S_AMF_ROOT}/amf")
    if(EXISTS "${D2S_AMF_ROOT}/amf/public/common/AMFFactory.cpp")
        target_sources(d2s_amd_encoder PRIVATE "${D2S_AMF_ROOT}/amf/public/common/AMFFactory.cpp")
    endif()
    if(EXISTS "${D2S_AMF_ROOT}/amf/public/common/Windows/UtilsWindows.cpp")
        target_sources(d2s_amd_encoder PRIVATE "${D2S_AMF_ROOT}/amf/public/common/Windows/UtilsWindows.cpp")
    endif()
endif()
set_target_properties(d2s_amd_encoder PROPERTIES OUTPUT_NAME "d2s_amd_encoder")
